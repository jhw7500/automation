def publish_approved_canary(approved, fleet, observe, action):
    import re

    def require(ok):
        if not ok:
            raise ValueError("approved canary identity changed; reconcile read-only")

    approved = dict(approved)
    require(approved["repository"] == "jhw7500/wlan-package")
    require(approved["base_ref"] == "master")
    require(approved["branch"] == "automation/common-workflows-v1.78.3")
    for key in ("base_sha", "head_sha", "tree_sha"):
        require(re.fullmatch(r"[0-9a-f]{40}", approved[key]) is not None)
    original = fleet._github_post, fleet.create_rollout_branch, fleet.create_pull_request
    branch_started = False
    pr_started = False

    def live(expected_head):
        current = observe()
        require(current == {"repository": approved["repository"], "base_ref": "master",
                            "base_sha": approved["base_sha"], "head_sha": expected_head})

    def branch(snapshot, name, *, commit):
        nonlocal branch_started
        require(not branch_started)
        require(snapshot.path.name == "wlan-package" and snapshot.default_branch == "master")
        require((snapshot.base_branch or snapshot.default_branch) == "master")
        require(snapshot.base_sha == commit.base_sha == approved["base_sha"])
        require(commit.head_sha == approved["head_sha"])
        require(commit.tree_sha == approved["tree_sha"] and name == approved["branch"])
        live(None)
        branch_started = True
        return original[1](snapshot, name, commit=commit)

    def post(repo, section, payload):
        require(branch_started and not pr_started and repo == "wlan-package")
        require(section in {"blobs", "trees", "commits", "refs"})
        if section == "refs":
            require(payload == {"ref": "refs/heads/" + approved["branch"],
                                "sha": approved["head_sha"]})
        live(None)
        return original[0](repo, section, payload)

    def pr(owner, repo, base, head, head_sha, title, body):
        nonlocal pr_started
        require(not pr_started)
        require((owner, repo, base, head, head_sha, title, body) ==
                ("jhw7500", "wlan-package", "master", approved["branch"], approved["head_sha"],
                 approved["title"], approved["body"]))
        live(approved["head_sha"])
        pr_started = True
        result = original[2](owner, repo, base, head, head_sha, title, body)
        live(approved["head_sha"])
        return result

    fleet._github_post, fleet.create_rollout_branch, fleet.create_pull_request = post, branch, pr
    try:
        return action()
    finally:
        fleet._github_post, fleet.create_rollout_branch, fleet.create_pull_request = original
